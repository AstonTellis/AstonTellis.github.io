Featured Queries
A2 — TOP CUSTOMERS BY BALANCE JOIN 
SELECT
  CONCAT(c.first_name, ' ', c.last_name) AS customer,
  c.customer_tier,
  COUNT(a.account_id)          AS accounts,
  SUM(a.balance)               AS total_balance
FROM customers c
JOIN accounts a ON c.customer_id = a.customer_id
WHERE a.is_active = 1
GROUP BY c.customer_id
ORDER BY total_balance DESC
LIMIT 10;
↳ Multi-account JOIN · GROUP BY · SUM · ORDER DESC
B2 — ANOMALY DETECTION Fraud 
SELECT t.transaction_id, t.amount,
  ROUND(avg_data.avg_spend, 2) AS cust_avg,
  ROUND(t.amount / avg_data.avg_spend, 2)
    AS multiple_of_average
FROM transactions t
JOIN accounts a ON t.account_id = a.account_id
JOIN (
  SELECT a2.customer_id,
         AVG(t2.amount) AS avg_spend
  FROM transactions t2
  JOIN accounts a2 ON t2.account_id = a2.account_id
  WHERE t2.transaction_type = 'Debit'
  GROUP BY a2.customer_id
) AS avg_data ON a.customer_id = avg_data.customer_id
WHERE t.amount > (avg_data.avg_spend * 2)
ORDER BY multiple_of_average DESC;
↳ Subquery · Personalised baseline · Fraud signal logic
C2 — MoM GROWTH (WINDOW FUNCTION) Advanced 
WITH monthly_totals AS (
  SELECT
    DATE_FORMAT(transaction_date,
      '%Y-%m')     AS month,
    ROUND(SUM(amount), 2) AS total_volume
  FROM transactions
  WHERE status = 'Completed'
  GROUP BY DATE_FORMAT(transaction_date, '%Y-%m')
)
SELECT month, total_volume,
  LAG(total_volume) OVER
    (ORDER BY month)  AS prev_month,
  ROUND((total_volume
    - LAG(total_volume) OVER
        (ORDER BY month))
    / LAG(total_volume) OVER
        (ORDER BY month) * 100, 1)
    AS mom_growth_pct
FROM monthly_totals;
↳ CTE · LAG() window function · Growth rate formula
B3 — RAPID SUCCESSION DETECTION Fraud 
SELECT
  t1.transaction_id     AS first_txn,
  t2.transaction_id     AS second_txn,
  TIMESTAMPDIFF(MINUTE,
    t1.transaction_date,
    t2.transaction_date) AS minutes_apart,
  t1.amount + t2.amount AS combined_value
FROM transactions t1
JOIN transactions t2
  ON  t1.account_id = t2.account_id
  AND t2.transaction_id > t1.transaction_id
  AND TIMESTAMPDIFF(MINUTE,
        t1.transaction_date,
        t2.transaction_date)
      BETWEEN 0 AND 10
WHERE t1.transaction_type = 'Debit'
ORDER BY combined_value DESC;
↳ Self-JOIN · TIMESTAMPDIFF · Card skimming pattern
